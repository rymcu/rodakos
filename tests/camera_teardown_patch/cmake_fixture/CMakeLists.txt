cmake_minimum_required(VERSION 3.16)
project(camera_overlay_replacement_fixture LANGUAGES C)
set(PROJECT_DIR "${RODAKOS_ROOT}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(video_dir "${RODAKOS_ROOT}/managed_components/espressif__esp_video")
set(sensor_dir "${RODAKOS_ROOT}/managed_components/espressif__esp_cam_sensor")
set(video_source "${video_dir}/src/device/esp_video_device_common.c")
set(sensor_source "${sensor_dir}/src/driver_dvp/esp_cam_ctlr_dvp_cam.c")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/unrelated.c" "void unrelated(void) {}\n")
foreach(component video sensor)
    set(inputs "${${component}_source}")
    if(FIXTURE_CASE STREQUAL "missing_${component}")
        set(inputs "${CMAKE_CURRENT_BINARY_DIR}/unrelated.c")
    elseif(FIXTURE_CASE STREQUAL "duplicate_${component}")
        list(APPEND inputs "${${component}_source}")
    endif()
    add_library(${component}_lib STATIC ${inputs})
endforeach()

function(idf_component_get_property output component property)
    if(component STREQUAL "espressif__esp_video")
        set(key video)
    elseif(component STREQUAL "espressif__esp_cam_sensor")
        set(key sensor)
    else()
        message(FATAL_ERROR "Unexpected component: ${component}")
    endif()
    if(property STREQUAL "COMPONENT_DIR")
        set(${output} "${${key}_dir}" PARENT_SCOPE)
    elseif(property STREQUAL "COMPONENT_LIB")
        set(${output} "${key}_lib" PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected component property: ${property}")
    endif()
endfunction()
function(idf_build_get_property output property)
    if(property STREQUAL "PYTHON")
        set(${output} "${Python3_EXECUTABLE}" PARENT_SCOPE)
    elseif(property STREQUAL "IDF_PATH")
        set(${output} "${RODAKOS_IDF_PATH}" PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected build property: ${property}")
    endif()
endfunction()

include("${RODAKOS_ROOT}/cmake/camera_teardown_patch.cmake")
foreach(component video sensor)
    get_target_property(compiled ${component}_lib SOURCES)
    set(expected "${camera_generated_${component}}")
    if(NOT compiled STREQUAL expected)
        message(FATAL_ERROR "Original source remains or generated source missing: ${compiled}")
    endif()
    get_target_property(includes ${component}_lib INCLUDE_DIRECTORIES)
    if(NOT "${RODAKOS_ROOT}/main/phone_os" IN_LIST includes)
        message(FATAL_ERROR "Private C ABI include missing")
    endif()
endforeach()
if(FIXTURE_CASE STREQUAL "fault")
    get_target_property(sensor_definitions sensor_lib COMPILE_DEFINITIONS)
    if(NOT "RODAKOS_CAMERA_DMA_FORCE_4096=1" IN_LIST sensor_definitions)
        message(FATAL_ERROR "Camera DMA fault compile definition missing")
    endif()
endif()
